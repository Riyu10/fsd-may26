const nam = document.querySelector('#name');
const age = document.querySelector('#age');
const address = document.querySelector('#address');
const clase = document.querySelector('#class');

// 1. Load saved values on page load
nam.value = localStorage.getItem('name') || '';
age.value = localStorage.getItem('age') || '';
address.value = localStorage.getItem('address') || '';
clase.value = localStorage.getItem('class') || '';

// 2. Save values on user input
const fields = [
  { el: nam, key: 'name' },
  { el: age, key: 'age' },
  { el: address, key: 'address' },
  { el: clase, key: 'class' },
];

fields.forEach(({ el, key }) => {
  el.addEventListener('input', () => {
    localStorage.setItem(key, el.value);
  });
});
